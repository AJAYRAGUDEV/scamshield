      // =========================================================================
      // INTERACTIVE PARTICLE NEURAL GRID BACKGROUND
      // =========================================================================
      function initBackgroundCanvas() {
        const canvas = el.bgCanvas;
        const ctx = canvas.getContext('2d');
        if (!ctx) return;

        let width = 0;
        let height = 0;
        let particles = [];
        const mouse = { x: -9999, y: -9999 };
        let sweepAngle = 0;

        function resize() {
          width = canvas.width = window.innerWidth;
          height = canvas.height = window.innerHeight;
          const count = Math.min(72, Math.max(28, Math.floor((width * height) / 22000)));
          particles = Array.from({ length: count }, () => ({
            x: Math.random() * width,
            y: Math.random() * height,
            vx: (Math.random() - 0.5) * 0.45,
            vy: (Math.random() - 0.5) * 0.45,
            r: Math.random() * 1.8 + 0.9,
          }));
        }

        window.addEventListener('resize', resize);
        window.addEventListener('mousemove', (e) => {
          mouse.x = e.clientX;
          mouse.y = e.clientY;
        });
        window.addEventListener('mouseleave', () => {
          mouse.x = -9999;
          mouse.y = -9999;
        });

        resize();

        function draw() {
          // Smoothly interpolate theme RGB
          for (let i = 0; i < 3; i++) {
            state.bgThemeRgb[i] += (state.targetThemeRgb[i] - state.bgThemeRgb[i]) * 0.04;
          }
          const [r, g, b] = state.bgThemeRgb.map((v) => Math.round(v));

          ctx.clearRect(0, 0, width, height);

          // Subtle radar rings near top-right
          const cx = width * 0.78;
          const cy = height * 0.28;
          sweepAngle += state.isScanning ? 0.035 : 0.008;
          ctx.save();
          ctx.strokeStyle = `rgba(${r}, ${g}, ${b}, 0.06)`;
          ctx.lineWidth = 1;
          [110, 210, 320].forEach((rad) => {
            ctx.beginPath();
            ctx.arc(cx, cy, rad, 0, Math.PI * 2);
            ctx.stroke();
          });
          // Sweep beam
          ctx.beginPath();
          ctx.moveTo(cx, cy);
          ctx.arc(cx, cy, 320, sweepAngle, sweepAngle + 0.35);
          ctx.closePath();
          const grad = ctx.createRadialGradient(cx, cy, 0, cx, cy, 320);
          grad.addColorStop(0, `rgba(${r}, ${g}, ${b}, 0.12)`);
          grad.addColorStop(1, `rgba(${r}, ${g}, ${b}, 0)`);
          ctx.fillStyle = grad;
          ctx.fill();
          ctx.restore();

          // Update & draw nodes
          const speedBoost = state.isScanning ? 2.4 : 1;
          for (let i = 0; i < particles.length; i++) {
            const p = particles[i];
            p.x += p.vx * speedBoost;
            p.y += p.vy * speedBoost;
            if (p.x < 0 || p.x > width) p.vx *= -1;
            if (p.y < 0 || p.y > height) p.vy *= -1;

            // Connect nearby nodes
            for (let j = i + 1; j < particles.length; j++) {
              const q = particles[j];
              const dx = p.x - q.x;
              const dy = p.y - q.y;
              const dist = Math.hypot(dx, dy);
              if (dist < 140) {
                const alpha = (1 - dist / 140) * 0.18;
                ctx.strokeStyle = `rgba(${r}, ${g}, ${b}, ${alpha})`;
                ctx.lineWidth = 1;
                ctx.beginPath();
                ctx.moveTo(p.x, p.y);
                ctx.lineTo(q.x, q.y);
                ctx.stroke();
              }
            }

            // Connect to pointer
            const md = Math.hypot(p.x - mouse.x, p.y - mouse.y);
            if (md < 170) {
              ctx.strokeStyle = `rgba(${r}, ${g}, ${b}, ${(1 - md / 170) * 0.35})`;
              ctx.lineWidth = 1.1;
              ctx.beginPath();
              ctx.moveTo(p.x, p.y);
              ctx.lineTo(mouse.x, mouse.y);
              ctx.stroke();
            }

            ctx.fillStyle = `rgba(${r}, ${g}, ${b}, 0.65)`;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
            ctx.fill();
          }

          requestAnimationFrame(draw);
        }

        requestAnimationFrame(draw);
      }

      // AUDIO SPECTRUM WAVEFORM CANVAS
      // =========================================================================
      function initAudioWaveCanvas() {
        const canvas = el.audioWaveCanvas;
        const ctx = canvas.getContext('2d');
        if (!ctx) return;

        let phase = 0;

        function renderWave() {
          const rect = canvas.getBoundingClientRect();
          if (rect.width > 0 && rect.height > 0) {
            if (canvas.width !== Math.floor(rect.width) || canvas.height !== Math.floor(rect.height)) {
              canvas.width = Math.floor(rect.width);
              canvas.height = Math.floor(rect.height);
            }
          }
          const w = canvas.width || 300;
          const h = canvas.height || 96;
          ctx.clearRect(0, 0, w, h);

          const bars = 44;
          const gap = 3;
          const barW = Math.max(2, (w - gap * (bars - 1)) / bars);
          const active = state.waveActive || state.isScanning;
          phase += active ? 0.12 : 0.03;

          const [r, g, b] = state.bgThemeRgb.map((v) => Math.round(v));

          for (let i = 0; i < bars; i++) {
            const norm = i / bars;
            const envelope = Math.sin(norm * Math.PI);
            const wave1 = Math.sin(phase + i * 0.42);
            const wave2 = Math.cos(phase * 1.4 - i * 0.25);
            const amp = active
              ? (0.25 + 0.72 * Math.abs(wave1 * 0.6 + wave2 * 0.4)) * envelope
              : (0.1 + 0.14 * Math.abs(wave1)) * (0.4 + 0.6 * envelope);
            const barH = Math.max(4, amp * (h - 16));
            const x = i * (barW + gap);
            const y = (h - barH) / 2;

            const grad = ctx.createLinearGradient(x, y, x, y + barH);
            grad.addColorStop(0, `rgba(${r}, ${g}, ${b}, 0.95)`);
            grad.addColorStop(1, 'rgba(59, 130, 246, 0.55)');
            ctx.fillStyle = grad;

            ctx.beginPath();
            ctx.roundRect(x, y, barW, barH, 3);
            ctx.fill();
          }

          requestAnimationFrame(renderWave);
        }

        requestAnimationFrame(renderWave);
      }

      // =========================================================================
      // LOCAL SMART SIMULATION ENGINE (MATCHES server.py SCHEMA 100%)
      // =========================================================================
      function runLocalTextEngine(text, offlineFlag = false) {
        const lower = text.toLowerCase();
        const scores = {};
        const flags = [];

        Object.entries(HEURISTIC_RULES).forEach(([key, cfg]) => {
          const found = cfg.phrases.filter((p) => lower.includes(p));
          scores[key] = Math.min(100, cfg.weight * found.length);
          found.forEach((p) => {
            flags.push(`Detected high-signal phrase "${p}" (${key.replace(/_/g, ' ')})`);
          });
        });

        // Additional subtle pattern boosts for realistic simulation
        if (/https?:\/\/|bit\.ly|tinyurl/i.test(text)) {
          scores.credential_payment_trap = Math.min(100, scores.credential_payment_trap + 20);
          flags.push('Contains an external link prompting urgent action');
        }
        if (/\b\d{4,6}\b/.test(text) && (lower.includes('otp') || lower.includes('code') || lower.includes('pin'))) {
          scores.credential_payment_trap = Math.min(100, scores.credential_payment_trap + 15);
        }

        const totalScore = Math.min(
          100,
          Math.round(
            scores.authority_impersonation * 0.35 +
            scores.urgency_isolation * 0.3 +
            scores.credential_payment_trap * 0.4 +
            scores.synthetic_or_script_signals * 0.25 +
            (flags.length >= 4 ? 18 : 0)
          )
        );

        const verdict = totalScore >= 60 ? 'High Risk' : totalScore >= 25 ? 'Suspicious' : 'Low Risk';
        const confidence = flags.length >= 3 ? 'High' : flags.length >= 1 ? 'Medium' : 'Medium';

        let scamType = 'No common scam pattern detected';
        if (scores.authority_impersonation >= 50 && scores.urgency_isolation >= 40) {
          scamType = 'Authority Impersonation / Digital Arrest Coercion';
        } else if (scores.credential_payment_trap >= 50) {
          scamType = 'Credential Phishing & Payment Trap';
        } else if (scores.synthetic_or_script_signals >= 20 && scores.credential_payment_trap >= 25) {
          scamType = 'Synthetic / Scripted Executive Wire Fraud';
        } else if (flags.length > 0) {
          scamType = 'Potential Social Engineering';
        }

        const explanation =
          verdict === 'High Risk'
            ? 'Multiple converging coercion indicators were identified—including authority pressure, time-sensitive isolation, or payment/credential demands. Treat this message as a high-probability scam attempt.'
            : verdict === 'Suspicious'
              ? 'Meaningful social-engineering signals appear in this message, though they are not conclusive on their own. Avoid acting until verified through an independent channel.'
              : 'Little evidence of common impersonation, urgency, or payment-trap tactics was detected in this text. Still exercise caution if the sender is unexpected.';

        const recommendedAction =
          verdict === 'High Risk'
            ? 'Do not transfer funds, click links, or share OTPs/PINs. End contact immediately and report the sender via your official bank or cybercrime portal.'
            : verdict === 'Suspicious'
              ? 'Pause and verify the claim independently using the organization\'s official app or phone number rather than replying directly.'
              : 'No urgent threat indicators found. Confirm any logistics directly inside the official service app if needed.';

        return {
          risk_score: totalScore,
          verdict,
          scam_type: scamType,
          confidence_level: offlineFlag ? 'Low' : confidence,
          signal_scores: {
            authority_impersonation: scores.authority_impersonation,
            urgency_isolation: scores.urgency_isolation,
            credential_payment_trap: scores.credential_payment_trap,
            synthetic_or_script_signals: scores.synthetic_or_script_signals,
          },
          red_flags: flags.slice(0, 20),
          explanation,
          recommended_action: recommendedAction,
          mode: 'text',
          offline_fallback_used: offlineFlag,
        };
      }

      // LIVE FASTAPI INTEGRATION (ONLY WHEN SERVED BY USER'S BACKEND)
      // =========================================================================
      function canCallSameOriginApi() {
        try {
          const proto = window.location.protocol;
          return (proto === 'https:' || proto === 'http:') && window.location.host !== '' && !window.location.href.startsWith('about:');
        } catch {
          return false;
        }
      }

      async function analyzeTextRequest(text) {
        if (state.engineMode === 'api' && canCallSameOriginApi()) {
          try {
            const fd = new FormData();
            fd.append('text', text);
            const key = el.apiKeyInput.value.trim();
            if (key) fd.append('api_key', key);

            const res = await fetch('/api/analyze-text', { method: 'POST', body: fd });
            const data = await res.json();
            if (!res.ok) {
              throw new Error(data.detail || `Server error (${res.status})`);
            }
            return data;
          } catch (err) {
            showToast(`Live API unavailable (${err.message}). Using local fallback.`, 'warn');
            return runLocalTextEngine(text, true);
          }
        }
        return runLocalTextEngine(text, false);
      }

      function animateCounter(element, target) {
        const start = parseInt(element.textContent, 10) || 0;
        const duration = 750;
        const startTime = performance.now();

        function tick(now) {
          const progress = Math.min(1, (now - startTime) / duration);
          const eased = 1 - Math.pow(1 - progress, 3);
          element.textContent = Math.round(start + (target - start) * eased);
          if (progress < 1) requestAnimationFrame(tick);
        }
        requestAnimationFrame(tick);
      }

      function escapeHtml(str) {
        return String(str)
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"/g, '&quot;')
          .replace(/'/g, '&#39;');
      }

      // =========================================================================
      // LIVE KEYWORD RADAR PREVIEW ON TEXT INPUT
      // =========================================================================
      function updateTextPreview() {
        const val = el.messageInput.value;
        el.charCounter.textContent = `${val.length.toLocaleString()} / ${MAX_TEXT_CHARS.toLocaleString()}`;

        const lower = val.toLowerCase();
        const hits = [];
        Object.values(HEURISTIC_RULES).forEach((cfg) => {
          cfg.phrases.forEach((p) => {
            if (lower.includes(p) && !hits.includes(p)) hits.push(p);
          });
        });

        if (hits.length === 0) {
          el.liveKeywordBar.classList.add('hidden');
          el.liveKeywordBar.classList.remove('flex');
          el.liveKeywordPills.innerHTML = '';
        } else {
          el.liveKeywordBar.classList.remove('hidden');
          el.liveKeywordBar.classList.add('flex');
          el.liveKeywordPills.innerHTML = hits
            .slice(0, 6)
            .map((h) => `<span class="px-2 py-0.5 rounded-md bg-rose-500/15 border border-rose-500/35 text-rose-300 font-mono">${escapeHtml(h)}</span>`)
            .join('');
        }
      }

      // =========================================================================
      // AUDIO SELECTION HANDLER
      // =========================================================================
      function handleAudioFile(file) {
        if (!file) return;
        const lowerName = file.name.toLowerCase();
        const validExt = ALLOWED_EXTENSIONS.some((ext) => lowerName.endsWith(ext));
        if (!validExt) {
          showToast('Unsupported audio format. Use MP3, WAV, M4A, OGG, or WEBM.', 'error');
          return;
        }
        if (file.size > MAX_AUDIO_BYTES) {
          showToast('Audio file exceeds the 20 MB limit.', 'error');
          return;
        }
        state.selectedAudioFile = file;
        state.selectedAudioSampleKey = null;
        el.selectedAudioName.textContent = file.name;
        el.selectedAudioSize.textContent = `(${(file.size / (1024 * 1024)).toFixed(2)} MB)`;
        el.selectedAudioBadge.classList.remove('hidden');

        try {
          el.audioPlayer.src = URL.createObjectURL(file);
        } catch {
          // Safe fallback if object URL unavailable
        }
        showToast(`Loaded audio: ${file.name}`, 'success');
      }

        function buildPromptForTask(taskKey, customGoal = '') {
          const msg = (el.messageInput.value || '').trim();
          const res = state.currentResult;
          const flagsText = res && res.red_flags && res.red_flags.length
            ? res.red_flags.map((f) => `- ${f}`).join('\n')
            : '- No specific red flags flagged yet';
          const quoteBlock = msg || '[Paste suspicious message here]';

          if (taskKey === 'system') {
            return {
              title: 'ScamShield AI — Backend System Prompt',
              body: SYSTEM_PROMPT_TEXT,
            };
          }
          if (taskKey === 'incident') {
            return {
              title: 'Incident & Cybercrime Report Prompt',
              body: `Draft a concise, factual incident report for my bank and cybercrime helpline based on this ScamShield AI assessment:\n\nVerdict: ${res ? res.verdict : 'Pending'} (Risk Score: ${res ? res.risk_score : 0}/100)\nSuspected Pattern: ${res ? res.scam_type : 'Unclassified'}\nObserved Red Flags:\n${flagsText}\n\nOriginal Message / Transcript:\n---\n${quoteBlock}\n---`,
            };
          }
          if (taskKey === 'safe_reply') {
            return {
              title: 'Safe Non-Escalating Verification Reply Prompt',
              body: `Write a short, polite, non-committal reply that refuses to share OTPs, PINs, or money and states that I will verify independently through official channels.\n\nContext (${res ? res.verdict : 'Unassessed'} — ${res ? res.scam_type : 'General'}):\n---\n${quoteBlock}\n---`,
            };
          }
          if (taskKey === 'custom' && customGoal) {
            return {
              title: `Custom Prompt — ${customGoal.slice(0, 42)}${customGoal.length > 42 ? '...' : ''}`,
              body: `Role: You are ScamShield AI, a cautious scam and impersonation intelligence assistant.\nTask: ${customGoal}\n\nCurrent Assessment: ${res ? `${res.verdict} (${res.risk_score}/100) — ${res.scam_type}` : 'Not yet scanned'}\nEvidence / Input:\n---\n${quoteBlock}\n---\n\nConstraints: Base your answer strictly on observable evidence, avoid alarmism, and end with one practical safe next step.`,
            };
          }
          return {
            title: 'Structured Scam Audit Prompt',
            body: `Analyze the following suspicious message or call transcript for authority impersonation, urgency/isolation, credential/payment traps, and synthetic/scripted signals. Return a calibrated risk score (0–100), concise red flags grounded in the text, and a safe next step:\n\n---\n${quoteBlock}\n---`,
          };
        }

        ta.select();

          document.execCommand('copy');
          showToast('Recommended action copied to clipboard', 'success');
        } catch {
          showToast('Unable to copy text automatically', 'error');
        }
        ta.remove();
      }
